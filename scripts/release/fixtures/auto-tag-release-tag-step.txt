          set -euo pipefail

          version="$(jq -r .version src-tauri/tauri.conf.json)"

          # Release builds only: reject anything that is not plain X.Y.Z semver
          # (no pre-release/build metadata), matching release.yml's `v*.*.*`.
          if ! [[ "$version" =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
            echo "::error::tauri.conf.json version '$version' is not X.Y.Z semver" >&2
            exit 1
          fi

          # src-tauri/Cargo.toml must agree with tauri.conf.json — a half-done
          # bump would ship a tag whose crate version disagrees with the app.
          # The read is section-aware and exits non-zero when it finds nothing
          # (#2169): a `version` outside [package] — [workspace.package] once
          # #2161 lands — or a quoted trailing comment must not be picked up,
          # because a wrong value here compares green against the wrong crate.
          # Regression suite: scripts/release/cargo-package-version.test.sh
          cargo_version="$(bash scripts/release/cargo-package-version.sh src-tauri/Cargo.toml)"
          if [ "$version" != "$cargo_version" ]; then
            echo "::error::version mismatch: tauri.conf.json=$version src-tauri/Cargo.toml=$cargo_version" >&2
            exit 1
          fi

          # #1439 P3-11 — package.json is the third source-of-truth version file
          # (versioning-and-artifacts.md). This workflow only fires on a
          # tauri.conf.json change, so a bump that forgot package.json would tag
          # a release whose npm/package version disagrees with the app.
          pkg_version="$(jq -r .version package.json)"
          if [ "$version" != "$pkg_version" ]; then
            echo "::error::version mismatch: tauri.conf.json=$version package.json=$pkg_version" >&2
            exit 1
          fi

          tag="v$version"

          # Idempotent no-op: tag already published for this version.
          if [ -n "$(git ls-remote --tags origin "$tag")" ]; then
            echo "Tag $tag already exists on origin — no-op."
            exit 0
          fi

          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git tag -a "$tag" -m "Release $tag"
          git push origin "$tag"
          echo "Pushed $tag — release.yml will build the draft release (manual publish)."
